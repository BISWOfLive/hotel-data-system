### 1.5 轻量 HTTP 探活 `probe_login_valid()`（免开浏览器）—— 逐字抄录

```python
def probe_login_valid(account: Optional[dict] = None, timeout_s: int = 30) -> bool:
    """轻量 HTTP 探测登录态（不开浏览器；ApiCollector 思路）。

    - 读登录态文件（account.storage_state_path > storage_states/<alias>.json > 旧 storage_state.json）
      拼 cookie header；
    - GET 经营报告首个 api（``collectors.rules.get_api_rules`` → ``pages['经营报告'].api_defs[0].url``），
      ``allow_redirects=False``；
    - 200 → True；301/302 且 Location/URL 含 login/passport/sso → False；401/403 → False；异常 → False；
    - 登录态缺失 → False；全局限频 sleep(Config.API_COLLECT_RATE_S)。
    """
    try:
        rules = get_api_rules()
        page_cfg = (rules.get("pages") or {}).get("经营报告") or {}
        api_defs = page_cfg.get("api_defs") or []
        if not api_defs or not api_defs[0].get("url"):
            return False
        url = api_defs[0]["url"]

        state_path = _resolve_state_path(account)
        if state_path is None or not state_path.exists():
            logger.info("登录态文件缺失，探测返回 False: {}", state_path)
            return False
        with open(state_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        cookies = data.get("cookies") or []
        if not isinstance(cookies, list) or not cookies:
            return False
        parts = [
            f"{c['name']}={c.get('value', '')}"
            for c in cookies
            if isinstance(c, dict) and c.get("name") is not None
        ]
        if not parts:
            return False
        cookie_header = "; ".join(parts)

        resp = requests.get(
            url,
            headers={"cookie": cookie_header, "user-agent": _USER_AGENT},
            allow_redirects=False,
            timeout=timeout_s,
        )
        status = int(resp.status_code)
        if status == 200:
            return True
        # 重定向且落到登录页 → 失效
        loc = (resp.headers.get("Location", "") or "").lower()
        redirect_haystack = f"{loc} {resp.url}".lower()
        if status in (301, 302) and any(m in redirect_haystack for m in LOGIN_URL_MARKERS):
            return False
        # 401/403 → 失效；其余状态保守视为未确认
        return False
    except Exception as exc:  # noqa: BLE001
        logger.warning("probe_login_valid 异常（按失效处理）: {}", exc)
        return False
    finally:
        time.sleep(Config.API_COLLECT_RATE_S)
```

### 1.6 `need_renewal()` —— 20 天阈值（精确实现与时间字段）

`crawl/login_manager.py:197-216`，逐字：

```python
    @staticmethod
    def need_renewal(last_login_at: Optional[str], max_age_days: int = 20) -> bool:
        """登录态是否需主动续登。

        None → True（从未登录）；超过 ``max_age_days`` 天 → True；
        解析失败 → False（保守：不主动续登，交由探测决定）。
        """
        if not last_login_at:
            return True
        try:
            dt = datetime.fromisoformat(str(last_login_at))
        except (TypeError, ValueError):
            return False
        now = datetime.now()
        if dt.tzinfo is not None:
            # dt 带时区：把 now 也归一化为本地 aware，避免 offset-naive/aware 相减抛错
            if now.tzinfo is None:
                now = now.astimezone()
        age = now - dt
        return age.total_seconds() > max_age_days * 86400
```

```python
            if not box or box.get("width", 0) <= 0:
                logger.warning("滑块无有效尺寸，无法拖拽（转人工）")
                return False

            start_y = box["y"] + box["height"] * 0.5
            target_x = box["x"] + box["width"] * 0.8

            mouse = page.mouse
            mouse.move(start_x, start_y)
            mouse.down()
            steps = random.randint(8, 14)
            step_px = (target_x - start_x) / steps
            for i in range(steps):
                forward = start_x + step_px * (i + 1) + random.uniform(-1, 2)
                mouse.move(forward, start_y)
                time.sleep(random.uniform(0.015, 0.040))
                if i == steps // 2:
                    # 回撤一次：模拟手抖
                    mouse.move(forward - random.uniform(2, 5), start_y)
                    time.sleep(random.uniform(0.015, 0.040))
            mouse.up()
            logger.info("滑块已自动拖动（人类轨迹模拟，{} 步）", steps)
            return True
```

```python
        """轮询等待用户（或验证码通过后）完成登录；每 5 秒打印诊断信息。"""
        logger.info(
            "→ 请在浏览器窗口内手动完成登录（如有滑块请手动拖动），最长等待 {} 秒…",
            timeout_s,
        )

        while time.time() < deadline:
            if self._is_logged_in(page):

                logger.info(
                    "仍在等待登录… URL={} | 标题={} | 页面内容={}",
                    page.url,
                    self._safe_title(page),
                    self._page_text_snippet(page, 80),
                )
            time.sleep(2)
        return False
```

```python
@contextmanager
def page_session(headless: Optional[bool] = None, storage_state: Optional[Path] = None,
                 save_state_to: Optional[Path] = None) -> Iterator[tuple[object, object, object]]:
    """一个"真人操作"会话：browser + context + page。

    - CDP 模式（连接用户自己的 Chrome）：复用用户已登录的 context，只开新标签页，
      结束时不关闭浏览器，仅关闭标签页并断开连接；
    - 独立模式：启动本机 Chrome/Chromium，结束自动关闭。
    返回 (browser, context, page)。若指定 save_state_to，退出时保存登录态。
    """
    p, browser, is_cdp = launch_browser(headless)
    if is_cdp:
        # 用户 Chrome：复用现有 context（共享登录 cookie），开一个新标签页
        ctx = browser.contexts[0] if browser.contexts else browser.new_context()
        page = ctx.new_page()
    else:
        ctx = new_context(browser, storage_state)
        page = ctx.new_page()
    try:
        yield browser, ctx, page
    finally:
        if save_state_to:
            try:
                ctx.storage_state(path=str(save_state_to))
                logger.info("登录态已保存: {}", save_state_to)
            except Exception as exc:  # noqa: BLE001
                logger.warning("保存登录态失败: {}", exc)
        try:
            page.close()
        except Exception:  # noqa: BLE001
            pass
        if not is_cdp:
            try:
                ctx.close()
            except Exception:  # noqa: BLE001
                pass
            try:
                browser.close()
            except Exception:  # noqa: BLE001
                pass
        try:
            p.stop()
        except Exception:  # noqa: BLE001
            pass
```

### 2.3 `launch_browser()`：CDP 优先 + `channel="chrome"` 回退链

`comparator/browsers.py:21-54`，逐字：

```python
def launch_browser(headless: Optional[bool] = None):
    """启动/连接浏览器，返回 (playwright, browser, is_cdp)。

    配置了 CHROME_CDP_URL → 优先连接用户已登录的 Chrome；
    CDP 连接失败 → 自动降级为独立启动（不再报错，登录/采集都能继续）。
    """
    p = sync_playwright().start()
    cdp = (Config.CHROME_CDP_URL or "").strip()
    if cdp:
        try:
            browser = p.chromium.connect_over_cdp(cdp, timeout=15000)
            logger.success("已连接用户 Chrome（CDP）: {} —— 使用你已登录的会话", cdp)
            return p, browser, True
        except Exception as exc:  # noqa: BLE001
            logger.warning("连接用户 Chrome 失败（{}）——自动降级为独立启动浏览器", exc)
    use_headless = Config.HOTEL_HEADLESS if headless is None else headless
    args = [
        "--disable-blink-features=AutomationControlled",
        "--disable-infobars",
        "--no-first-run",
        "--no-default-browser-check",
    ]
    try:
        browser = p.chromium.launch(
            channel="chrome",
            headless=use_headless,
            args=args,
        )
        logger.info("已使用本机 Chrome（headless={}）", use_headless)
        return p, browser, False
    except Exception as exc:  # noqa: BLE001
        logger.warning("未找到本机 Chrome（{}），回退到 Playwright Chromium", exc)
        browser = p.chromium.launch(headless=use_headless, args=args)
        return p, browser, False
```

### 4.2 鼠标轨迹 `human_move_mouse()`

```python
def human_move_mouse(page, tx: float, ty: float, steps: int = 12) -> None:
    """鼠标从随机起点，经抖动轨迹移动到目标点（模拟人手）。"""
    try:
        vp = page.viewport_size or {"width": 1280, "height": 800}
        sx = random.randint(0, max(1, int(vp["width"] * 0.6)))
        sy = random.randint(0, max(1, int(vp["height"] * 0.4)))
        page.mouse.move(sx, sy)
        time.sleep(rand(0.05, 0.2))
        steps = max(6, steps)
        for i in range(1, steps + 1):
            t = i / steps
            ease = t * t * (3 - 2 * t)
            x = sx + (tx - sx) * ease + random.uniform(-6, 6)
            y = sy + (ty - sy) * ease + random.uniform(-6, 6)
            page.mouse.move(x, y)
            time.sleep(rand(0.02, 0.06))
        page.mouse.move(tx + random.uniform(-2, 2), ty + random.uniform(-2, 2))
    except Exception:  # noqa: BLE001
        try:
            page.mouse.move(tx, ty)
        except Exception:  # noqa: BLE001
            pass
```

### 4.7 `wait_for()` 的精确实现

```python
def wait_for(page, condition, timeout: float = 20.0, interval: float = 0.8, desc: str = "") -> bool:
    """带真人节奏的轮询等待；超时返回 False（不抛异常）。"""
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            if condition():
                return True
        except Exception:  # noqa: BLE001
            pass
        time.sleep(interval * rand(0.7, 1.3))
    logger.warning("等待超时（{}s）: {}", timeout, desc or "条件未满足")
    return False
```
